def create
  file = params[:file]
  if file.present?
    data = JSON.parse(file.read)
    import_data(data)
    flash[:notice] = "Dados importados com sucesso!"
  else
    flash[:alert] = "Nenhum arquivo selecionado."
  end
  redirect_to gerenciamento_index_url
end
